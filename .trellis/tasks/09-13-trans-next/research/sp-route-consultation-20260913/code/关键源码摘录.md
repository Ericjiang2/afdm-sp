  285     def generate_uwa_channel(self):
  286         """
  287         Generate MATLAB-style UWA acoustic channel parameters.
  288
  289         Port of `acousticChannelEnhanced` from MATLAB:
  290           1. Exponential delay increments with mean `tau_mean`
  291           2. Exponential power-delay profile with attenuation `att_dB`
  292              over total delay spread `tau_spread`
  293           3. Rayleigh path amplitudes + random phases, norm-normalized
  294           4. Non-uniform Doppler scaling factors a_i = v_i / c_sound with
  295              per-path Gaussian velocity v_i ~ |N(v_mean, v_std^2)|
  296           5. Doppler-corrected discrete delay indices delay_idx_bar
  297
  298         Returns
  299         -------
  300         delay_bar     : (P,) float   — corrected path delays in seconds
  301         gain_tap      : (P,) complex — complex path gains, ||gain_tap||_2 = 1
  302         delay_idx_bar : (P,) int     — corrected 1-based sample delay indices
  303         path_var      : (P,) float   — exponential PDP variances before fading
  304         doppler_a     : (P,) float   — Doppler scaling factor a_i = v_i/c_sound
  305         velocities    : (P,) float   — per-path velocities in m/s
  306         resample_attempts : int      — accepted draw count (1 = accepted immediately)
  307         """
  308         P = int(self.P_paths)
  309         max_supported = self.max_delay_idx_bar_supported
  310
  311         for attempt in range(1, self.uwa_resample_max_tries + 1):
  312             # 1) Delay distribution (MATLAB: makedist('exponential','mu',tau_mean))
  313             d_tau = np.random.exponential(scale=self.tau_mean, size=P)
  314             d_tau_index = np.maximum(1, np.ceil(d_tau / self.Ts).astype(int))
  315             delay_index = np.cumsum(d_tau_index)
  316             delay_vec = (delay_index - 1) * self.Ts
  317
  318             # 2) Gain power model (exponential decay over delay spread)
  319             alpha = np.log(10 ** (self.att_dB / 10.0)) / self.tau_spread
  320             path_var = np.exp(-alpha * delay_vec)
  321
  322             # MATLAB: sigma = sqrt(path_var * 2/(4-pi)); amp = raylrnd(sigma)
  323             sigma = np.sqrt(path_var * 2.0 / (4.0 - np.pi))
  324             amp = np.random.rayleigh(scale=sigma)
  325             amp_norm = np.linalg.norm(amp)
  326             if amp_norm > 0:
  327                 amp = amp / amp_norm
  328
  329             phases = 2.0 * np.pi * np.random.rand(P)
  330             gain_tap = amp * np.exp(1j * phases)
  331
  332             # 3) Non-uniform Doppler scaling
  333             velocities = np.abs(self.v_mean + self.v_std * np.random.randn(P))
  334             doppler_a = velocities / self.c_sound
  335
  336             # 4) Delay correction under Doppler scaling
  337             delay_idx_bar = np.maximum(
  338                 1,
  339                 np.ceil(delay_index / (1.0 + doppler_a)).astype(int)
  340             )
  341
  342             # Deep-learning-friendly calibration:
  343             # Reject rare draws whose corrected delay spread exceeds the AFDM
  344             # support window. This preserves the MATLAB stochastic law while
  345             # avoiding systematic clipping that would distort labels/features.
  346             if np.max(delay_idx_bar) <= max_supported:
  347                 delay_bar = (delay_idx_bar - 1) * self.Ts
  348                 return (
  349                     delay_bar, gain_tap, delay_idx_bar, path_var,
  350                     doppler_a, velocities, attempt
  351                 )
  352
  353         # Fallback: if all attempts overflow, clip once and continue.
  354         delay_idx_bar = np.minimum(delay_idx_bar, max_supported)
  355         delay_bar = (delay_idx_bar - 1) * self.Ts

  448     def gen_h_daft(self,
  449                    chan_coef: np.ndarray,
  450                    delay_index: np.ndarray,
  451                    doppler_index: np.ndarray) -> np.ndarray:
  452         """Build the N×N DAFT-domain channel matrix from integer tap parameters.
  453
  454         Uses the Python DAFT convention (Bemani Eq. 3.39):
  455
  456             H[m, n] += h_i * exp(j*2π * [ -c1*l_i² + c2*(m²-n²) - l_i*n/N ])
  457
  458         where  m = (n + δ_i) mod N,  δ_i = l_i * K_factor + k_i,
  459         and  K_factor = round(2 * N * c1).
  460
  461         IMPORTANT: This is the PYTHON convention (signs of c1, c2 terms differ
  462         from MATLAB GEN_H_DAFT which uses +c1*l² and c2=0).  The Python chirp
  463         matrix Lambda_c = diag(exp(+j*2π*c*n²)) produces a different analytical
  464         formula than MATLAB's convention.
  465
  466         Args:
  467             chan_coef:     1-D complex array of length P (tap gains)
  468             delay_index:   1-D int array of length P (delay bin indices l_i ≥ 0)
  469             doppler_index: 1-D int array of length P (Doppler bin indices k_i)
  470
  471         Returns:
  472             H: (N_total × N_total) complex channel matrix in DAFT domain.
  473         """
  474         N, c1 = self.N_total, self.c1
  475         convention = getattr(self, 'gen_h_daft_convention', 'python')
  476         c2 = 0.0 if convention == 'matlab' else self.c2
  477         K_factor = max(int(np.round(2 * N * c1)), 1)
  478         H = np.zeros((N, N), dtype=complex)
  479         n_arr = np.arange(N, dtype=np.float64)
  480         for i in range(len(chan_coef)):
  481             l_i = int(delay_index[i])
  482             k_i = int(doppler_index[i])
  483             if convention == 'matlab':
  484                 q_arr = n_arr
  485                 loc_val = k_i + 2.0 * N * c1 * l_i
  486                 p_arr = np.mod(np.round(q_arr - loc_val).astype(int), N)
  487                 phase = (2.0 * np.pi / N) * (
  488                     N * c1 * (l_i ** 2)
  489                     - q_arr * l_i
  490                     + N * c2 * (q_arr ** 2 - p_arr.astype(np.float64) ** 2)
  491                 )
  492                 H[p_arr, np.arange(N)] += chan_coef[i] * np.exp(1j * phase)
  493             else:
  494                 delta = l_i * K_factor + k_i
  495                 m_arr = ((n_arr.astype(int) + delta) % N)
  496                 m_arr_f = m_arr.astype(np.float64)
  497                 phase = (2.0 * np.pi) * (
  498                     -c1 * (l_i ** 2)
  499                     + c2 * (m_arr_f ** 2 - n_arr ** 2)
  500                     - float(l_i) * n_arr / N
  501                 )
  502                 H[m_arr, np.arange(N)] += chan_coef[i] * np.exp(1j * phase)
  503         return H

  505     def generate_channel_matrix(self, max_speed_kmh=300, channel_type='EVA'):
  506         """
  507         Generate AFDM channel matrix using the MATLAB UWA acoustic model.
  508
  509         Backward compatibility:
  510         - `max_speed_kmh` is accepted but ignored
  511         - `channel_type` is accepted but ignored
  512
  513         Physical model ported from MATLAB `AFDM_UWA_IDMA.m`:
  514           1. Draw UWA path delays / gains / Doppler-scaling factors
  515           2. Map physical delays to AFDM delay bins via `self.delay_resolution`
  516           3. Map MATLAB normalized Doppler D_i = a_i * fc / fs to AFDM bins via
  517              k_i = round(D_i * N_total), then clip to system design cap
  518           4. Build H_time = Σ_i g_i · Δ(k_i) · Π^{l_i}
  519           5. Normalize Frobenius norm to sqrt(N_total), matching MATLAB
  520           6. Compute H_eff = A @ H_time @ A^H
  521
  522         Returns:
  523             H_eff: Effective channel matrix in DAFT domain (N_total x N_total)
  524             H_time: Time-domain channel matrix (N_total x N_total)
  525             channel_params: Dictionary of channel parameters
  526         """
  527         delays, g_i_raw, delay_idx_bar, path_var, doppler_a, velocities, resample_attempts = \
  528             self.generate_uwa_channel()
  529         taps = len(g_i_raw)
  530
  531         # Delay mapping: physical UWA delays → AFDM delay bins
  532         l_i = np.round(delays / self.delay_resolution).astype(int)
  533         l_i = np.clip(l_i, 0, self.l_max_cap)
  534         collision = getattr(self, 'delay_collision', None)
  535         if collision == 'neighbor':
  536             if taps < 2:
  537                 raise ValueError('neighbor delay collision requires at least two paths')
  538             l_i = np.array(l_i, dtype=int, copy=True)
  539             l_i[1] = int(np.clip(int(l_i[0]) + 1, 0, self.l_max_cap))
  540             if l_i[1] == l_i[0]:
  541                 l_i[0] = int(np.clip(int(l_i[1]) - 1, 0, self.l_max_cap))
  542             if l_i[0] == l_i[1]:
  543                 raise ValueError('neighbor delay collision could not be realized')
  544         elif collision not in (None,):
  545             raise ValueError('unknown delay_collision')
  546
  547         # ── Doppler: physical continuous Doppler → AFDM bin indices ──────────
  548         # D_i = a_i * fc / fs  (MATLAB normalized Doppler, dimensionless)
  549         # k_real_i = D_i * N_total  (un-rounded, continuous fractional bin index)
  550         D_vals  = doppler_a * self.fc / self.fs
  551         k_real  = D_vals * self.N_total
  552         k_clip  = min(self.k_max_cap,
  553                       int(np.floor(self.N_total / 2) - self.k_guard - 1))
  554         k_real  = np.clip(k_real, -float(k_clip), float(k_clip))
  555         k_max   = float(np.max(np.abs(k_real))) if taps > 0 else 0.0
  556
  557         # ── Doppler quantization: integer or fractional (Dirichlet) mode ─────
  558         #
  559         # INTEGER MODE  (self.use_frac_doppler = False):
  560         #   Round each physical k_real to the nearest integer bin.
  561         #
  562         # FRACTIONAL MODE  (self.use_frac_doppler = True,  default):
  563         #   Each physical path is decomposed into (2*k_leak + 1) virtual integer-
  564         #   bin taps via the Dirichlet kernel (mirrors MATLAB AFDM_UWA_acoustic.m
  565         #   with USE_FRAC_DOPPLER=true, K_LEAK=5).  This creates a sparse integer-
  566         #   indexed representation that is compatible with GEN_H_DAFT / gen_h_daft
  567         #   and the downstream OMP / joint-MP channel estimators.
  568         if not self.use_frac_doppler:
  569             # ── Integer mode ──────────────────────────────────────────────────
  570             k_int_all = np.round(k_real).astype(int)
  571             k_int_all = np.clip(k_int_all, -k_clip, k_clip)
  572             chan_coef_exp = list(g_i_raw)
  573             l_exp         = list(l_i.astype(int))
  574             k_exp         = list(k_int_all)
  575             k_max_int     = int(np.max(np.abs(k_int_all))) if taps > 0 else 0
  576             # Per-path k_i stored as integer for channel_params
  577             k_i_stored = k_int_all.astype(float)
  578         else:
  579             # ── Fractional (Dirichlet decomposition) mode ─────────────────────
  580             # Each physical path i → 2*k_leak+1 virtual integer taps, where
  581             #   virtual_k = k_base + m,   m ∈ {-k_leak, …, +k_leak}
  582             #   virtual_gain = g_i * dirichlet_alpha(m, kappa, N),  energy-normalised
  583             chan_coef_exp = []
  584             l_exp         = []
  585             k_exp         = []
  586             m_range       = np.arange(-self.k_leak, self.k_leak + 1, dtype=int)
  587             prune_thres   = self.frac_prune_rel_thres
  588             for i in range(taps):
  589                 k_base = int(np.round(k_real[i]))
  590                 kappa  = k_real[i] - k_base      # fractional residual |kappa| ≤ 0.5
  591                 # Dirichlet weights for this path
  592                 a_vec  = np.array(
  593                     [self.dirichlet_alpha(int(m), kappa, self.N_total) for m in m_range],
  594                     dtype=complex
  595                 )
  596                 # Energy-normalise (matches MATLAB: a_vec = a_vec / sqrt(sum(|a_vec|^2)))
  597                 a_norm = np.sqrt(np.sum(np.abs(a_vec) ** 2) + 1e-300)
  598                 a_vec /= a_norm
  599                 # Amplitude pruning: drop weak Dirichlet sidelobes
  600                 # (mirrors MATLAB FRAC_PRUNE_REL_THRES = 0.01)
  601                 a_pow = np.abs(a_vec) ** 2
  602                 a_peak = np.max(a_pow)
  603                 for mm, m in enumerate(m_range):
  604                     k_eff = k_base + int(m)
  605                     if abs(k_eff) > k_clip:
  606                         continue   # discard out-of-range virtual taps
  607                     if a_pow[mm] < prune_thres * a_peak:
  608                         continue   # prune weak Dirichlet sidelobe
  609                     chan_coef_exp.append(g_i_raw[i] * a_vec[mm])
  610                     l_exp.append(int(l_i[i]))
  611                     k_exp.append(k_eff)
  612             k_max_int   = int(max((abs(k) for k in k_exp), default=0))
  613             k_i_stored  = k_real   # keep physical fractional value for diagnostics
  614
  615         P_actual = len(chan_coef_exp)
  616         self._P_actual = P_actual  # store for OMP sparsity
  617
  618         # ── Build H_eff via gen_h_daft (analytical DAFT formula) ──────────────
  619         # gen_h_daft requires integer delay + Doppler indices and produces the
  620         # DAFT-domain (H_eff) matrix directly from the expanded virtual taps.
  621         chan_coef_arr = np.array(chan_coef_exp, dtype=complex)
  622         l_arr         = np.array(l_exp,         dtype=int)
  623         k_arr         = np.array(k_exp,         dtype=int)
  624
  625         if P_actual > 0:
  626             H_eff_unnorm = self.gen_h_daft(chan_coef_arr, l_arr, k_arr)
  627         else:
  628             H_eff_unnorm = np.zeros((self.N_total, self.N_total), dtype=complex)
  629
  630         # Frobenius normalisation: ||H_eff||_F = sqrt(N)  (matches MATLAB)
  631         H_fro = np.linalg.norm(H_eff_unnorm, 'fro')
  632         if H_fro > 0:
  633             scale          = np.sqrt(float(self.N_total)) / H_fro
  634             H_eff          = H_eff_unnorm * scale
  635             chan_coef_arr *= scale
  636         else:
  637             scale = 1.0
  638             H_eff = H_eff_unnorm
  639
  640         # Back-compute normalised physical path gains (for diagnostics / RL reward)
  641         g_i  = g_i_raw * scale
  642         # H_time (time-domain): invert DAFT transform for backward compatibility
  643         # H_time = A^H @ H_eff @ A
  644         H_time = self._A_H @ H_eff @ self._A
  645
  646         channel_params = {
  647             'g_i':              g_i,
  648             'g_i_raw':          g_i_raw,
  649             'l_i':              l_i,
  650             'k_i':              k_i_stored,
  651             'k_real':           k_real,           # continuous physical Doppler bins
  652             'k_max':            k_max,
  653             'k_max_int':        k_max_int,
  654             'delay_spread':     int(np.max(l_i)) if taps > 0 else 0,
  655             'taps':             taps,
  656             'P_actual':         P_actual,         # expanded virtual tap count
  657             'chan_coef_exp':     chan_coef_arr,    # expanded (normalised) gains
  658             'l_exp':            l_arr,            # expanded delay indices
  659             'k_exp':            k_arr,            # expanded integer Doppler indices
  660             'delay_idx_bar':    delay_idx_bar,
  661             'delay_seconds':    delays,
  662             'path_var':         path_var,
  663             'doppler_a':        doppler_a,
  664             'velocities_ms':    velocities,
  665             'resample_attempts':resample_attempts,
  666             'fro_scale':        scale,
  667             'use_frac_doppler': self.use_frac_doppler,

  669             'channel_model':    'uwa_acoustic',
  670             'c1':               self.c1,
  671             'c2':               self.c2,
  672         }

   42 def construct_sensing_matrix(N, x_pilot_scaled, c1, c2, l_max, k_max):
   43     """
   44     Build AFDM OMP sensing matrix Φ ∈ ℂ^{N × Q}.
   45
   46     For a single unit-gain path at delay l, integer Doppler k, the expected
   47     DAFT-domain received signal at sub-carrier q is:
   48
   49         Φ[q, col(l,k)] = exp( j2π(−c1·l² + c2·(q²−p²) − l·p/N) ) · x_p[p]
   50         where  p = (q − δ) mod N,  δ = l·K + k,  K = 2·k_max + 1.
   51
   52     Column ordering:
   53         col(l, k) = l * K + (k + k_max)          (k in [−k_max, +k_max])
   54
   55     Parameters
   56     ----------
   57     N               : int    — total DAFT-domain subcarriers
   58     x_pilot_scaled  : (N,)   — pilot frame  (e.g. sqrt(rho) * zc_seq)
   59     c1, c2          : float  — AFDM chirp parameters
   60     l_max           : int    — maximum delay tap index   (0 … l_max)
   61     k_max           : int    — maximum Doppler index     (−k_max … +k_max)
   62
   63     Returns
   64     -------
   65     Phi : (N, Q) complex ndarray,   Q = (l_max+1) * (2*k_max+1)
   66     """
   67     K = 2 * k_max + 1                            # columns per delay row
   68     Q = (l_max + 1) * K                          # total dictionary columns
   69     q_arr = np.arange(N, dtype=np.float64)       # row indices (all N)
   70
   71     Phi = np.zeros((N, Q), dtype=complex)
   72
   73     col = 0
   74     for l in range(l_max + 1):
   75         l_sq = float(l) ** 2
   76         for k in range(-k_max, k_max + 1):
   77             delta = l * K + k                    # DAFT shift for this (l, k)
   78             p_arr = (q_arr - delta) % N          # input sub-carrier indices (N,)
   79             phase = np.exp(1j * 2.0 * np.pi * (
   80                 -c1 * l_sq
   81                 + c2 * (q_arr ** 2 - p_arr ** 2)
   82                 - float(l) * p_arr / N
   83             ))
   84             Phi[:, col] = phase * x_pilot_scaled[p_arr.astype(int)]
   85             col += 1

   94 def omp_sparse(y, Phi, sparsity):
   95     """
   96     Standard Orthogonal Matching Pursuit (OMP).
   97
   98     Solves:  min ‖y − Φ·h‖₂   s.t.  ‖h‖₀ ≤ sparsity
   99
  100     Parameters
  101     ----------
  102     y         : (N,) complex — observation vector
  103     Phi       : (N, Q) complex — sensing / dictionary matrix
  104     sparsity  : int — maximum number of non-zeros to recover
  105
  106     Returns
  107     -------
  108     h_est   : (Q,) complex — sparse coefficient vector (zeros outside support)
  109     support : list[int]    — column indices selected
  110     """
  111     Q = Phi.shape[1]
  112     residual = y.copy()
  113     support = []
  114     Phi_H = Phi.conj().T          # precomputed conjugate-transpose
  115
  116     for _ in range(sparsity):
  117         correlations = np.abs(Phi_H @ residual)     # (Q,) correlation magnitudes
  118         # Suppress already-selected columns
  119         for idx in support:
  120             correlations[idx] = 0.0
  121         new_idx = int(np.argmax(correlations))
  122         support.append(new_idx)
  123
  124         # Least-squares on current support
  125         Phi_S = Phi[:, support]                     # (N, s)
  126         gains_S, _, _, _ = np.linalg.lstsq(Phi_S, y, rcond=None)
  127
  128         # Update residual
  129         residual = y - Phi_S @ gains_S
  130
  131         # Early stop: relative residual power negligible
  132         if np.linalg.norm(residual) < 1e-10 * np.linalg.norm(y):
  133             break
  134
  135     # Final LS refinement on full support
  136     h_est = np.zeros(Q, dtype=complex)
  137     if support:
  138         Phi_S = Phi[:, support]
  139         gains_final, _, _, _ = np.linalg.lstsq(Phi_S, y, rcond=None)
  140         for i, idx in enumerate(support):
  141             h_est[idx] = gains_final[i]
  142
  143     return h_est, support
